-- View all users with roles
SELECT u.id, u.username, u.enabled, 
       STRING_AGG(r.role, ', ') AS roles
FROM users u
LEFT JOIN user_roles r ON u.id = r.user_id
GROUP BY u.id, u.username, u.enabled
ORDER BY u.id;

-- View all employees
SELECT * FROM emp_table ORDER BY id;

-- View all jobs with employee names
SELECT j.id, j.title, j.description, j.min_salary, 
       e.first_name, e.last_name
FROM job j
LEFT JOIN emp_table e ON j.employee_id = e.id
ORDER BY j.id;

-- View all projects with employee names
SELECT p.id, p.project_name, p.project_description, p.budget,
       e.first_name, e.last_name
FROM project_table p
LEFT JOIN emp_table e ON p.employee_id = e.id
ORDER BY p.id;

-- Count summary
SELECT 'Users' AS Table_Name, COUNT(*) AS Record_Count FROM users
UNION ALL
SELECT 'User Roles', COUNT(*) FROM user_roles
UNION ALL
SELECT 'Employees', COUNT(*) FROM emp_table
UNION ALL
SELECT 'Jobs', COUNT(*) FROM job
UNION ALL
SELECT 'Projects', COUNT(*) FROM project_table;
